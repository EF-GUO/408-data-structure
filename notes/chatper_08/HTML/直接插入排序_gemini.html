void InsertSort(ElemType A[], int n){
int i, j;
for(i=2; i<=n; i++) // 依次插入
if(A[i] < A[i-1]){ // 若小于前驱
A[0] = A[i]; // 复制为哨兵
for(j=i-1; A[0]<A[j]; --j) // 从后往前查
A[j+1] = A[j]; // 向后挪位
A[j+1] = A[0]; // 复制到位置
}
}
