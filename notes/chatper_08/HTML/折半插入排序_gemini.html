折半查找教材实现

void InsertSort(ElemType A[], int n){
int i, j, low, high, mid;
for(i=2; i<=n; i++){
A[0] = A[i]; // 暂存到哨兵
low = 1; high = i-1; // 初始化查找范围
while(low <= high){ // 折半查找
mid = (low + high) / 2;
if(A[mid] > A[0])
high = mid - 1; // 左半区
else
low = mid + 1; // 右半区(含相等)
}
for(j=i-1; j>=high+1; --j) // 统一后移
A[j+1] = A[j];
A[high+1] = A[0]; // 插入位置
}
}
