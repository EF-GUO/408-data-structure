void BubbleSort(ElemType A[], int n){
int i, j;
bool flag;
for(i=0; i<n-1; i++){ // 进行 n-1 趟冒泡
flag = false; // 表示本趟是否发生交换
for(j=0; j<n-i-1; j++){ // 一趟冒泡过程
if(A[j] > A[j+1]){ // 若逆序
swap(A[j], A[j+1]); // 交换元素
flag = true; // 标记发生交换
}
}
if(flag == false) // 若本趟未交换
return; // 说明表已经有序，提前结束
}
}
